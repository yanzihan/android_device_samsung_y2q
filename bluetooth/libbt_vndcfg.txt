# Broadcom BCM4375 build-time configuration (hardware/broadcom/libbt).
#
# Consumed at BUILD time by gen-buildcfg.sh, which turns each surviving line into
# "#define <line, first = replaced by a space>".  Two rules follow from that:
# names are the library's macro names, values are quoted to become C strings, and
# the file must contain NO blank lines - the script's blank-line filter does not
# work, so one becomes a bare "#define " and the build fails with "macro name
# missing".
#
# Selected from device.mk via soong_config_set(brcm_libbt, custom_bt_config, ...).
# The runtime counterpart is bt_vendor.conf in this directory.
#
# Qualcomm's msm_geni_serial registers the BT UART (qupv3_se6_4uart) as
# /dev/ttyHS0.  Do not copy y2s's /dev/ttySAC1 - that tree is Exynos.
BLUETOOTH_UART_DEVICE_PORT = "/dev/ttyHS0"
FW_PATCHFILE_LOCATION = "/vendor/firmware"
FW_PATCH_SETTLEMENT_DELAY_MS = 0
UART_TARGET_BAUD_RATE = 3000000
LPM_IDLE_TIMEOUT_MULTIPLE = 5
SCO_USE_I2S_INTERFACE = TRUE
SCO_I2SPCM_IF_ROLE = 0
# Drive the controller's sleep/wake through the kernel's /proc/bluetooth/sleep
# nodes.  With the library default (FALSE) the body of upio_set() for
# UPIO_LPM_MODE/UPIO_BT_WAKE is compiled out, so the chip can be put to sleep and
# never woken and Bluetooth cannot start.  The two node paths are the library's
# own defaults and the kernel provides them.
BT_WAKE_VIA_PROC = TRUE
# conf.c's default is /etc/bluetooth/bt_vendor.conf, and /etc resolves to
# /system/etc here, which the Bluetooth HAL domain may not read.
VENDOR_LIB_CONF_FILE = "/vendor/etc/bluetooth/bt_vendor.conf"
